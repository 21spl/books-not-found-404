@Entity
@Table(
    name = "inventory_reservations",
    indexes = {
        @Index(name = "idx_book_expiry", columnList = "book_id, expires_at")
    }
)
@NoArgConstructor
@Getter
public class InventoryReservation {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "book_id", nullable = false)
    private Book book;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "user_id", nullable = false)
    private User user;

    @Column(name = "quantity", nullable = false)
    private int quantity;
    
    @Column(name = "quantity", nullable = false)
    private double totalValue;

    @Column(name = "expires_at", nullable = false)
    private LocalDateTime expires_at;
    
    
    private static final int minutesToHold = 10;

    // Default Constructor, Getters, and Setters
    public InventoryReservation() {}

    public InventoryReservation(Book book, User user, int quantity) {
        this.book = book;
        this.user = user;
        this.quantity = quantity;
        this.expires_at = LocalDateTime.now().plusMinutes(minutesToHold);
        this.totalValue = calculateTotalValue();
    }

    public Long getId() { return id; }
    public Book getBook() { return book; }
    public User getUser() { return user; }
    public int getQuantity() { return quantity; }
    public LocalDateTime getExpiresAt() { return expires_at; }
    
    public void setExpiresAt(LocalDateTime expiresAt) { this.expires_at = expiresAt; }
    
    public double calculateTotalValue(){
    		int bookPrice = this.book.getPrice();
    		int quantity = this.quantity;
    		double total = 0;
    		total+= (bookPrice * quantity);
    }
    
    public void setTotalValue(double totalValue){
    		this.totalValue = totalValue;
    }
}

public interface InventoryReservationRepository extends JpaRepository<InventoryReservation, Long> {
    

    // calculates total reserved quantity for a book that haven't expired yet
    @Query("SELECT COALESCE(SUM(r.quantity), 0) FROM InventoryReservation r " +
            "WHERE r.book.id = :bookId AND r.expiresAt > :now")
    int getActiveReservedQuantity(@Param("bookId") Long bookId, @Param("now") LocalDateTime now);

    // bulk delete expired reservations
    @Modifying 
    @Query("DELETE FROM InventoryReservation r WHERE r.expiresAt < :now")
    void deleteExpiredReservations(@Param("now") LocalDateTime now);

    // find active reservation for a specific user and book (for payment retries)
    Optional<InventoryReservation> findByBookIdAndUserIdAndExpiresAtAfter(Long bookId, Long userId, 
        LocalDateTime now);

    // find the list of active reservation for a specific user 
    List<InventoryReservation> findByUserIdAndExpiresAtAfter(Long userId, LocalDateTime now);
}

public class InventoryReservationService{

    private final UserService userService;
    private final BookRepository bookRepository;
    private final InventoryReservationRepository reservationRepository;


    @Transactional 
    public boolean reserveInventory(Long bookId, Long userId, int quantity){

        // 1. Lock the Book row directly from the repo. 
        // This transaction explicitly holds the PESSIMISTIC_WRITE lock.
        Book book = bookRepository.findByIdWithPessimisticWrite(bookId)
            .orElseThrow(() -> new ResourceNotFoundException("Book not found with ID: " + bookId));

        // 2. Delegate user lookup to its legitimate domain service
        User user = userService.findUserById(userId);
        if(user == null){
            throw new UserNotFoundException("User not found with ID: " + userId);
        }

        // 3. Evaluate active reservations 
        int preReserved = reservationRepository.getActiveReservedQuantity(bookId, LocalDateTime.now());
        int realAvailableStock = book.getCopies() - preReserved;

        // 4. Secure reservation if stock matches

        try{
            CheckoutSummaryDTO summary = checkoutService.getCheckoutSummary(authentication.getName());

            model.addAttribute("cartItems", summary.getCartItems());
            model.addAttribute("totalAmount", summary.getTotalAmount());

            if (!summary.getStockWarnings().isEmpty()) {
                model.addAttribute("outOfStockWarnings", summary.getStockWarnings());
                return "customer/cart";
            }
            return "customer/checkout";

        }catch(UserNotFoundException e){
            return "redirect:/auth/login";
        } catch (EmptyCartException e) {
            return "redirect:/cart";
        } catch(NotEnoughStockException e){
            return "redirect:/cart";
        }
        
   }


   @PostMapping("/placeOrder")
    public String placeOrder(
            @Valid @ModelAttribute("orderForm") OrderFormDto form, 
            BindingResult bindingResult,                       
            Authentication authentication, 
            Model model) {

        // If validations failed, skip business logic and return the error view immediately
        if (bindingResult.hasErrors()) {
            CheckoutSummaryDTO summary = checkoutService.getCheckoutSummary(authentication.getName());
            model.addAttribute("cartItems", summary.getCartItems());
            model.addAttribute("totalAmount", summary.getTotalAmount());
            return "customer/checkout";
        }

        try {
            checkoutService.initiateCheckoutPayment(authentication.getName(), form.getAddress(), form.getPhone());
            return "redirect:/checkout/payment_page";
        } catch (OutOfStockException e) {
            return "customer/checkout";
        }
    }
